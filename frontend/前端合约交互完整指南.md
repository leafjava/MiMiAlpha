## 🔧 Step 1: 创建合约 Hook

### 文件：`src/hooks/useContract.ts`

```typescript
import { useEffect, useState } from 'react';
import { ethers } from 'ethers';
import contractAddresses from '../contracts/contract-addresses.json';
import MockERC20ABI from '../contracts/MockERC20.json';
import SmartFacilitatorABI from '../contracts/SmartFacilitator.json';

export function useContract() {
  const [provider, setProvider] = useState<ethers.BrowserProvider | null>(null);
  const [signer, setSigner] = useState<ethers.Signer | null>(null);
  const [account, setAccount] = useState<string>('');
  const [chainId, setChainId] = useState<number>(0);

  // 初始化 Provider
  useEffect(() => {
    if (typeof window.ethereum !== 'undefined') {
      const browserProvider = new ethers.BrowserProvider(window.ethereum);
      setProvider(browserProvider);

      // 监听账户变化
      window.ethereum.on('accountsChanged', (accounts: string[]) => {
        if (accounts.length > 0) {
          setAccount(accounts[0]);
        } else {
          setAccount('');
        }
      });

      // 监听网络变化
      window.ethereum.on('chainChanged', (chainId: string) => {
        setChainId(parseInt(chainId, 16));
        window.location.reload();
      });
    }
  }, []);

  // 连接钱包
  const connectWallet = async () => {
    if (!provider) {
      throw new Error('No provider found');
    }

    try {
      const accounts = await provider.send('eth_requestAccounts', []);
      const signer = await provider.getSigner();
      const network = await provider.getNetwork();

      setAccount(accounts[0]);
      setSigner(signer);
      setChainId(Number(network.chainId));

      return accounts[0];
    } catch (error) {
      console.error('Failed to connect wallet:', error);
      throw error;
    }
  };

  // 切换到 Hardhat 本地网络
  const switchToHardhat = async () => {
    if (!window.ethereum) {
      throw new Error('No wallet found');
    }

    try {
      await window.ethereum.request({
        method: 'wallet_switchEthereumChain',
        params: [{ chainId: '0x539' }], // 1337 的十六进制
      });
    } catch (switchError: any) {
      // 如果网络不存在，添加它
      if (switchError.code === 4902) {
        await window.ethereum.request({
          method: 'wallet_addEthereumChain',
          params: [{
            chainId: '0x539',
            chainName: 'Hardhat Local',
            rpcUrls: ['http://127.0.0.1:8545'],
            nativeCurrency: {
              name: 'Ethereum',
              symbol: 'ETH',
              decimals: 18
            }
          }]
        });
      } else {
        throw switchError;
      }
    }
  };

  // 获取 Token 合约
  const getTokenContract = () => {
    if (!signer) throw new Error('No signer');
    return new ethers.Contract(
      contractAddresses.contracts.MockERC20,
      MockERC20ABI.abi,
      signer
    );
  };

  // 获取 SmartFacilitator 合约
  const getFacilitatorContract = () => {
    if (!signer) throw new Error('No signer');
    return new ethers.Contract(
      contractAddresses.contracts.SmartFacilitator,
      SmartFacilitatorABI.abi,
      signer
    );
  };

  return {
    provider,
    signer,
    account,
    chainId,
    connectWallet,
    switchToHardhat,
    getTokenContract,
    getFacilitatorContract,
    isConnected: !!account,
    isHardhatNetwork: chainId === 1337,
  };
}
```

## 🤖 Step 2: 创建 SmartFacilitator Hook

### 文件：`src/hooks/useSmartFacilitator.ts`

```typescript
import { useState, useCallback } from 'react';
import { ethers } from 'ethers';
import { useContract } from './useContract';

export interface AgentInfo {
  owner: string;
  name: string;
  balance: bigint;
  totalSpent: bigint;
  transactionCount: bigint;
  active: boolean;
  creditScore: bigint;
}

export interface QuotaUsage {
  dailySpent: bigint;
  dailyLimit: bigint;
  monthlySpent: bigint;
  monthlyLimit: bigint;
}

export function useSmartFacilitator() {
  const { getFacilitatorContract, getTokenContract, account } = useContract();
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  // 创建 AI Agent
  const createAgent = useCallback(async (agentAddress: string, agentName: string) => {
    setLoading(true);
    setError(null);

    try {
      const facilitator = getFacilitatorContract();
      const tx = await facilitator.createAgent(agentAddress, agentName);
      await tx.wait();

      return tx.hash;
    } catch (err: any) {
      setError(err.message);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [getFacilitatorContract]);

  // 为 Agent 充值
  const depositFunds = useCallback(async (agentAddress: string, amount: string) => {
    setLoading(true);
    setError(null);

    try {
      const token = getTokenContract();
      const facilitator = getFacilitatorContract();
      const amountWei = ethers.parseEther(amount);

      // 1. 授权
      const approveTx = await token.approve(
        await facilitator.getAddress(),
        amountWei
      );
      await approveTx.wait();

      // 2. 充值
      const depositTx = await facilitator.depositFunds(agentAddress, amountWei);
      await depositTx.wait();

      return depositTx.hash;
    } catch (err: any) {
      setError(err.message);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [getFacilitatorContract, getTokenContract]);

  // 执行支付
  const executePayment = useCallback(async (
    agentAddress: string,
    recipientAddress: string,
    amount: string,
    service: string
  ) => {
    setLoading(true);
    setError(null);

    try {
      const facilitator = getFacilitatorContract();
      const amountWei = ethers.parseEther(amount);

      const tx = await facilitator.executePayment(
        agentAddress,
        recipientAddress,
        amountWei,
        service
      );
      const receipt = await tx.wait();

      return {
        hash: tx.hash,
        success: true,
        receipt
      };
    } catch (err: any) {
      setError(err.message);
      throw err;
    } finally {
      setLoading(false);
    }
  }, [getFacilitatorContract]);

  // 获取 Agent 信息
  const getAgentInfo = useCallback(async (agentAddress: string): Promise<AgentInfo> => {
    try {
      const facilitator = getFacilitatorContract();
      const info = await facilitator.getAgentInfo(agentAddress);

      return {
        owner: info.owner,
        name: info.name,
        balance: info.balance,
        totalSpent: info.totalSpent,
        transactionCount: info.transactionCount,
        active: info.active,
        creditScore: info.creditScore
      };
    } catch (err: any) {
      setError(err.message);
      throw err;
    }
  }, [getFacilitatorContract]);

  // 获取配额使用情况
  const getQuotaUsage = useCallback(async (agentAddress: string): Promise<QuotaUsage> => {
    try {
      const facilitator = getFacilitatorContract();
      const quota = await facilitator.getQuotaUsage(agentAddress);

      return {
        dailySpent: quota.dailySpent,
        dailyLimit: quota.dailyLimit,
        monthlySpent: quota.monthlySpent,
        monthlyLimit: quota.monthlyLimit
      };
    } catch (err: any) {
      setError(err.message);
      throw err;
    }
  }, [getFacilitatorContract]);

  // 获取统计信息
  const getStats = useCallback(async () => {
    try {
      const facilitator = getFacilitatorContract();
      const [totalAgents, totalPayments] = await Promise.all([
        facilitator.totalAgents(),
        facilitator.totalPayments()
      ]);

      return {
        totalAgents: Number(totalAgents),
        totalPayments: Number(totalPayments)
      };
    } catch (err: any) {
      setError(err.message);
      throw err;
    }
  }, [getFacilitatorContract]);

  return {
    createAgent,
    depositFunds,
    executePayment,
    getAgentInfo,
    getQuotaUsage,
    getStats,
    loading,
    error
  };
}
```

## 💰 Step 3: 创建 Token Hook

### 文件：`src/hooks/useToken.ts`

```typescript
import { useState, useCallback, useEffect } from 'react';
import { ethers } from 'ethers';
import { useContract } from './useContract';

export function useToken() {
  const { getTokenContract, account } = useContract();
  const [balance, setBalance] = useState<string>('0');
  const [loading, setLoading] = useState(false);

  // 获取余额
  const getBalance = useCallback(async (address?: string) => {
    try {
      const token = getTokenContract();
      const targetAddress = address || account;
      if (!targetAddress) return '0';

      const balance = await token.balanceOf(targetAddress);
      const formatted = ethers.formatEther(balance);
      setBalance(formatted);
      return formatted;
    } catch (err) {
      console.error('Failed to get balance:', err);
      return '0';
    }
  }, [getTokenContract, account]);

  // 转账
  const transfer = useCallback(async (to: string, amount: string) => {
    setLoading(true);
    try {
      const token = getTokenContract();
      const amountWei = ethers.parseEther(amount);
      const tx = await token.transfer(to, amountWei);
      await tx.wait();
      
      // 更新余额
      await getBalance();
      
      return tx.hash;
    } catch (err: any) {
      throw err;
    } finally {
      setLoading(false);
    }
  }, [getTokenContract, getBalance]);

  // 授权
  const approve = useCallback(async (spender: string, amount: string) => {
    setLoading(true);
    try {
      const token = getTokenContract();
      const amountWei = ethers.parseEther(amount);
      const tx = await token.approve(spender, amountWei);
      await tx.wait();
      return tx.hash;
    } catch (err: any) {
      throw err;
    } finally {
      setLoading(false);
    }
  }, [getTokenContract]);

  // 自动更新余额
  useEffect(() => {
    if (account) {
      getBalance();
      // 每 10 秒更新一次
      const interval = setInterval(getBalance, 10000);
      return () => clearInterval(interval);
    }
  }, [account, getBalance]);

  return {
    balance,
    getBalance,
    transfer,
    approve,
    loading
  };
}
```

## 🎨 Step 4: 在 SubscriptionMarket 中集成

### 修改：`src/components/SubscriptionMarket.tsx`

在现有的 SubscriptionMarket 组件中添加合约交互：

```typescript
import { useContract } from '../hooks/useContract';
import { useSmartFacilitator } from '../hooks/useSmartFacilitator';
import { useToken } from '../hooks/useToken';

export function SubscriptionMarket() {
  const { account, connectWallet, isConnected, isHardhatNetwork, switchToHardhat } = useContract();
  const { executePayment, loading: facilitatorLoading } = useSmartFacilitator();
  const { balance, transfer } = useToken();

  // 租赁订阅
  const handleRent = async (subscription: any) => {
    if (!isConnected) {
      await connectWallet();
      return;
    }

    if (!isHardhatNetwork) {
      await switchToHardhat();
      return;
    }

    try {
      // 1. 创建 Agent（如果还没有）
      const agentAddress = account; // 使用用户地址作为 Agent 地址

      // 2. 执行支付
      const txHash = await executePayment(
        agentAddress,
        subscription.owner,
        subscription.price.toString(),
        subscription.service
      );

      console.log('Payment successful:', txHash);
      alert('租赁成功！');
    } catch (error) {
      console.error('Rent failed:', error);
      alert('租赁失败：' + error.message);
    }
  };

  return (
    <div className="subscription-market">
      {/* 显示连接状态 */}
      <div className="wallet-status">
        {isConnected ? (
          <>
            <div>账户: {account.slice(0, 6)}...{account.slice(-4)}</div>
            <div>余额: {balance} cUSD</div>
            {!isHardhatNetwork && (
              <button onClick={switchToHardhat}>
                切换到 Hardhat 网络
              </button>
            )}
          </>
        ) : (
          <button onClick={connectWallet}>连接钱包</button>
        )}
      </div>

      {/* 订阅列表 */}
      <div className="subscriptions">
        {subscriptions.map(sub => (
          <div key={sub.id} className="subscription-card">
            <h3>{sub.service}</h3>
            <p>价格: {sub.price} cUSD</p>
            <button 
              onClick={() => handleRent(sub)}
              disabled={facilitatorLoading}
            >
              {facilitatorLoading ? '处理中...' : '租赁'}
            </button>
          </div>
        ))}
      </div>
    </div>
  );
}
```

## 📝 Step 5: 创建 Agent 管理组件

### 新文件：`src/components/AgentManager.tsx`

```typescript
import { useState, useEffect } from 'react';
import { useContract } from '../hooks/useContract';
import { useSmartFacilitator, AgentInfo } from '../hooks/useSmartFacilitator';
import { ethers } from 'ethers';

export function AgentManager() {
  const { account, isConnected } = useContract();
  const { 
    createAgent, 
    getAgentInfo, 
    depositFunds, 
    getQuotaUsage,
    loading 
  } = useSmartFacilitator();

  const [agentInfo, setAgentInfo] = useState<AgentInfo | null>(null);
  const [quotaUsage, setQuotaUsage] = useState<any>(null);
  const [agentName, setAgentName] = useState('');
  const [depositAmount, setDepositAmount] = useState('');

  // 加载 Agent 信息
  useEffect(() => {
    if (account) {
      loadAgentInfo();
    }
  }, [account]);

  const loadAgentInfo = async () => {
    try {
      const info = await getAgentInfo(account);
      setAgentInfo(info);

      if (info.active) {
        const quota = await getQuotaUsage(account);
        setQuotaUsage(quota);
      }
    } catch (error) {
      console.log('Agent not found or error:', error);
    }
  };

  const handleCreateAgent = async () => {
    try {
      await createAgent(account, agentName);
      alert('Agent 创建成功！');
      await loadAgentInfo();
    } catch (error) {
      alert('创建失败：' + error.message);
    }
  };

  const handleDeposit = async () => {
    try {
      await depositFunds(account, depositAmount);
      alert('充值成功！');
      await loadAgentInfo();
    } catch (error) {
      alert('充值失败：' + error.message);
    }
  };

  if (!isConnected) {
    return <div>请先连接钱包</div>;
  }

  if (!agentInfo || !agentInfo.active) {
    return (
      <div className="agent-manager">
        <h2>🤖 创建 AI Agent</h2>
        <p>基于 AINFT Agent Framework 构建</p>
        
        <input
          type="text"
          placeholder="Agent 名称"
          value={agentName}
          onChange={(e) => setAgentName(e.target.value)}
        />
        
        <button onClick={handleCreateAgent} disabled={loading}>
          {loading ? '创建中...' : '创建 Agent'}
        </button>
      </div>
    );
  }

  return (
    <div className="agent-manager">
      <h2>🤖 AI Agent 管理</h2>
      
      {/* Agent 信息 */}
      <div className="agent-info">
        <h3>{agentInfo.name}</h3>
        <p>地址: {account}</p>
        <p>余额: {ethers.formatEther(agentInfo.balance)} cUSD</p>
        <p>累计消费: {ethers.formatEther(agentInfo.totalSpent)} cUSD</p>
        <p>交易次数: {agentInfo.transactionCount.toString()}</p>
        <p>信用评分: {agentInfo.creditScore.toString()} / 1000</p>
      </div>

      {/* 配额使用情况 */}
      {quotaUsage && (
        <div className="quota-usage">
          <h3>📊 配额使用情况</h3>
          <p>日限额: {ethers.formatUnits(quotaUsage.dailyLimit, 6)} USDT</p>
          <p>日已用: {ethers.formatUnits(quotaUsage.dailySpent, 6)} USDT</p>
          <p>月限额: {ethers.formatUnits(quotaUsage.monthlyLimit, 6)} USDT</p>
          <p>月已用: {ethers.formatUnits(quotaUsage.monthlySpent, 6)} USDT</p>
        </div>
      )}

      {/* 充值 */}
      <div className="deposit">
        <h3>💰 充值</h3>
        <input
          type="number"
          placeholder="充值金额"
          value={depositAmount}
          onChange={(e) => setDepositAmount(e.target.value)}
        />
        <button onClick={handleDeposit} disabled={loading}>
          {loading ? '充值中...' : '充值'}
        </button>
      </div>
    </div>
  );
}
```

## 🐛 常见问题

### Q1: 钱包连接失败
**解决：** 确保安装了 MetaMask 或 OKX 钱包

### Q2: 网络不匹配
**解决：** 使用 `switchToHardhat()` 切换到本地网络

**解决：** 检查余额是否足够，确保已授权
